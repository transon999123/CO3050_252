\subsection*{2. Phân công công việc chi tiết}

\subsubsection*{2.1. Nguyễn Phúc Khánh}
\begin{itemize}
    \item \textbf{Nhiệm vụ chính:}
    \begin{itemize}
        \item Soạn thảo phần \textbf{Giới thiệu} và \textbf{Thiết kế hệ thống} trong báo cáo.
        \item Phát triển code Verilog để xử lý ảnh trên Vivado.
        \item Viết tài liệu kỹ thuật mô tả module Verilog.
    \end{itemize}
    \item \textbf{Lịch trình cụ thể:}
    \begin{itemize}
        \item 26/9 -- 2/10: Đọc và tìm hiểu các đề tài.
        \item 3/10 -- 5/10: Cùng nhóm phân chia công việc.
        \item 7/10 -- 12/10: Viết code Verilog và mô phỏng trên Vivado.
        \item 13/10 -- 14/10: Kiểm tra và tích hợp với script Python.
        \item 15/10 -- 16/10: Hoàn thiện phần báo cáo được phân công.
        \item 20/12 -- 31/12: Mở rộng kích thước ảnh có thể xử lý
        \item 31/12 -- 2/1: Hoàn thiện báo cáo cuối kỳ
    \end{itemize}
\end{itemize}

\subsubsection*{2.2. Lương Gia Minh}
\begin{itemize}
    \item \textbf{Nhiệm vụ chính:}
    \begin{itemize}
        \item Soạn thảo phần \textbf{Kiến thức nền tảng} và \textbf{Kết quả mô phỏng} trong báo cáo.
        \item Hoàn thành script Python để chuyển đổi ảnh sang định dạng \texttt{.hex}.
    \end{itemize}
    \item \textbf{Lịch trình cụ thể:}
    \begin{itemize}
        \item 26/9 -- 2/10: Đọc và tìm hiểu các đề tài.
        \item 3/10 -- 5/10: Cùng nhóm phân chia công việc.
        \item 7/10 -- 12/10: Hoàn thành module chuyển đổi ảnh sang file \texttt{.hex} bằng Python.
        \item 13/10 -- 14/10: Kiểm tra và tích hợp với code Verilog.
        \item 15/10 -- 16/10: Hoàn thiện phần báo cáo được phân công
        \item 31/12 -- 2/1: Hoàn thiện báo cáo cuối kỳ
    \end{itemize}
\end{itemize}
