% Định nghĩa macro vẽ Venn để dùng lại cho từng bài
% Cú pháp: \venn{Label S}{Label P}{Label M}{Shading Code}{X Mark Code}
% Lệnh vẽ sơ đồ Venn chuẩn
% #1: Nhãn S, #2: Nhãn P, #3: Nhãn M
% #4: Code tô màu (Shading)
% #5: Code đánh dấu X
\newcommand{\venn}[5]{
    \begin{center}
    \begin{tikzpicture}[scale=0.7]
        \coordinate (S) at (0,0);
        \coordinate (P) at (0:2.5cm);
        \coordinate (M) at (60:2.5cm);
        \def\R{1.8cm}
        \begin{scope} #4 \end{scope}
        \draw (S) circle (\R) node[below left] {S (#1)};
        \draw (P) circle (\R) node[below right] {P (#2)};
        \draw (M) circle (\R) node[above] {M (#3)};
        \node at (-1.5, -1.0) {\textbf{S}};
        \node at (4.0, -1.0) {\textbf{P}};
        \node at (1.25, 3.2) {\textbf{M}};
        #5
    \end{tikzpicture}
    \end{center}
}

% Định nghĩa lệnh vẽ Sơ đồ Venn 3 vòng tròn
% Cú pháp: \VennDiagram{ShadeOptions}{TextOptions}{Caption}
% ShadeOptions: Các lệnh \fill hoặc \pattern để tô màu vùng
% TextOptions: Các lệnh \node để đặt dấu X hoặc văn bản
\newcommand{\VennDiagram}[3]{
    \begin{center}
    \begin{tikzpicture}[scale=0.8]
        % Định nghĩa tọa độ tâm
        \coordinate (S) at (0,0);
        \coordinate (P) at (3,0);
        \coordinate (M) at (1.5, 2.6);

        % Tô màu (Shading) - Được truyền vào từ tham số #1
        \begin{scope}
            \fill[white] (-2,-2) rectangle (5,5); % Nền trắng
            #1
        \end{scope}

        % Vẽ viền 3 vòng tròn
        \draw[thick] (S) circle (1.8cm) node[below left] {S (Tiểu từ)};
        \draw[thick] (P) circle (1.8cm) node[below right] {P (Đại từ)};
        \draw[thick] (M) circle (1.8cm) node[above] {M (Trung từ)};
        
        % Nhãn S, P, M cho các vòng tròn
        \node at (-1.2, 1.5) {}; 
        
        % Các ký hiệu bổ sung (dấu X) - Được truyền vào từ tham số #2
        #2
        
    \end{tikzpicture}
    \\ \textbf{#3}
    \end{center}
}
